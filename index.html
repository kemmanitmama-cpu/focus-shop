from pathlib import Path from PIL import Image import zipfile, shutil, re # ============================== # ตั้งค่าไฟล์ # ============================== zip_in = Path("/mnt/data/FOCUS_SHOP_catalog.zip") shirt_src = Path("/mnt/data/04BA8608-8538-44B7-BAC2-C0053EE6E35C.jpeg") work = Path("/mnt/data/FOCUS_SHOP_catalog_updated") if work.exists(): shutil.rmtree(work) work.mkdir(parents=True) # ============================== # แตกไฟล์เว็บเดิม # ============================== with zipfile.ZipFile(zip_in, "r") as z: z.extractall(work) # ============================== # เพิ่มรูปเสื้อ # ============================== assets = work / "assets" assets.mkdir(exist_ok=True) shutil.copy2( shirt_src, assets / "jacket_added.jpg" ) # ============================== # แก้ไข index.html # ============================== index = work / "index.html" html = index.read_text(encoding="utf-8") # เปลี่ยนรูปเสื้อเดิมเป็นรูปใหม่ html = html.replace( 'src="assets/jacket.jpg"', 'src="assets/jacket_added.jpg"' ) # ============================== # ลบข้อความ "ราคาเพชร" # ============================== patterns = [ r'<[^>]*>ราคาเพชร[^<]*]*>', r'ราคาเพชร\s*[:：]?\s*[\d,]+\s*(?:เพชร|บาท)?', r'ราคาเพชร\s*[:：]?\s*[\d,.]+', ] for pattern in patterns: html = re.sub( pattern, '', html, flags=re.IGNORECASE ) # ============================== # เปลี่ยนชื่อให้เหลือ "ราคาสินค้า" # ============================== html = html.replace( "ราคาเพชร", "ราคาสินค้า" ) html = html.replace( "ราคาเพชร:", "ราคาสินค้า:" ) html = html.replace( "ราคาเพชร :", "ราคาสินค้า :" ) # ============================== # ลบช่องกรอกราคาเพชร # ============================== html = re.sub( r']*>ราคาเพชร.*?', '', html, flags=re.IGNORECASE | re.DOTALL ) html = re.sub( r']*(?:ราคาเพชร|diamond)[^>]*>', '', html, flags=re.IGNORECASE ) # ============================== # บันทึก HTML # ============================== index.write_text( html, encoding="utf-8" ) # ============================== # สร้าง ZIP ใหม่ # ============================== zip_out = Path( "/mnt/data/FOCUS_SHOP_catalog_updated.zip" ) if zip_out.exists(): zip_out.unlink() with zipfile.ZipFile( zip_out, "w", zipfile.ZIP_DEFLATED ) as z: for p in work.rglob("*"): if p.is_file(): z.write( p, p.relative_to(work) ) print("================================") print("อัปเดต FOCUS SHOP เรียบร้อย") print("================================") print("ลบราคาเพชรแล้ว") print("เหลือเฉพาะราคาสินค้า") print("เพิ่มรูปเสื้อแล้ว") print(f"ไฟล์ ZIP: {zip_out}")
